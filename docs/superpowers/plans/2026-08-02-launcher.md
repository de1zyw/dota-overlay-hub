# Pre-launch Checklist Launcher Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A `launcher.py` GUI, in the overlay's own dark-gradient visual style, that checks the environment (deps, Steam/Dota, GSI cfg, server_log.txt, local Steam account), reports problems without auto-fixing them, and launches `app.py` as a separate process once ready.

**Architecture:** A pure-logic `CHECKS` list of `(label, check_fn) -> (severity, message)` tuples, no Qt dependency, consumed by a `LauncherWindow(QWidget)` that reuses `overlay_window._GradientPanel` for background styling. Launch spawns `app.py` via `subprocess.Popen` and quits the launcher.

**Tech Stack:** Python stdlib (`importlib.util`, `os`, `subprocess`, `sys`) + PyQt6 (already a project dependency).

## Global Constraints

- No automated tests for this project (standing decision) — manual verification only, same as every prior task.
- Report-only checks: launcher never installs/copies/modifies anything.
- Linux only for now — no Windows-specific paths or packaging.
- Launch button disabled iff any check reports `ERROR`; `WARNING` never blocks launch.
- Visual style must reuse `overlay_window._GradientPanel` and `config.py`'s existing color constants (`COLOR_GREEN`, `COLOR_RED`) for consistency with the overlay.

### Task 1: Check functions

**Files:**
- Create: `launcher_checks.py`

**Interfaces:**
- Produces:

  - `CHECKS: list[tuple[str, callable]]` — each callable takes no args, returns `(status, message)` where `status` is one of the three constants above and `message` is a short user-facing string (Russian, matching the rest of the UI's language).

- [ ] **Step 1: Write `launcher_checks.py`**

```python
"""Pure pre-launch environment checks for launcher.py - no Qt dependency,
so they can be run/verified directly from a plain interpreter. Report-only:
each check just returns a severity + message, never modifies anything."""
import importlib.util
import os

import config

STATUS_OK = "ok"
STATUS_WARN = "warn"
STATUS_ERROR = "error"


def check_dependencies():
    missing = [m for m in ("PyQt6", "requests", "pynput") if importlib.util.find_spec(m) is None]
    if missing:
        return STATUS_ERROR, f"Не установлены зависимости: {', '.join(missing)} (pip install -r requirements.txt --break-system-packages)"
    return STATUS_OK, "Все зависимости установлены"


def check_dota_found():
    dota_dir = os.path.dirname(config.SERVER_LOG_PATH)
    if os.path.isdir(dota_dir):
        return STATUS_OK, "Dota 2 найдена"
    return STATUS_WARN, f"Папка Dota 2 не найдена: {dota_dir}"


def check_gsi_cfg():
    cfg_path = os.path.join(config.GSI_CFG_DIR, "gamestate_integration_dota_overlay.cfg")
    if os.path.isfile(cfg_path):
        return STATUS_OK, "GSI-конфиг установлен"
    return STATUS_WARN, f"GSI-конфиг не найден в {config.GSI_CFG_DIR} — скопируй gamestate_integration_dota_overlay.cfg туда (live-пик работать не будет, остальная стата — будет)"


def check_server_log():
    if os.path.isfile(config.SERVER_LOG_PATH):
        return STATUS_OK, "server_log.txt найден"
    return STATUS_WARN, "server_log.txt ещё не создан — появится после первого принятого матча"


def check_steam_account():
    if config.MY_ACCOUNT_ID is not None:
        return STATUS_OK, f"Steam-аккаунт определён (account_id={config.MY_ACCOUNT_ID})"
    return STATUS_WARN, "Steam-аккаунт не определён — подсветка «это ты» работать не будет"


CHECKS = [
    ("Python-зависимости", check_dependencies),
    ("Steam/Dota 2 на диске", check_dota_found),
    ("GSI-конфиг", check_gsi_cfg),
    ("server_log.txt", check_server_log),
    ("Steam-аккаунт", check_steam_account),
]
```

```python
"""Pre-launch checklist window: runs launcher_checks.CHECKS, shows status
in the overlay's own dark-gradient style, and launches app.py as a
separate process once ready. Report-only - never auto-fixes anything."""
import os
import subprocess
import sys

from PyQt6.QtCore import Qt
from PyQt6.QtWidgets import (
    QApplication,
    QHBoxLayout,
    QLabel,
    QPushButton,
    QVBoxLayout,
    QWidget,
)

import config
from launcher_checks import CHECKS, STATUS_ERROR, STATUS_OK, STATUS_WARN
from overlay_window import _GradientPanel

_STATUS_COLOR = {
    STATUS_OK: config.COLOR_GREEN,
    STATUS_WARN: "#e8b339",
    STATUS_ERROR: config.COLOR_RED,
}
_STATUS_ICON = {STATUS_OK: "✓", STATUS_WARN: "!", STATUS_ERROR: "✗"}

PROJECT_DIR = os.path.dirname(os.path.abspath(__file__))


def _check_item(label, status, message):
    item = QWidget()
    layout = QVBoxLayout(item)
    layout.setContentsMargins(0, 4, 0, 4)
    layout.setSpacing(2)

    header = QWidget()
    header_layout = QHBoxLayout(header)
    header_layout.setContentsMargins(0, 0, 0, 0)
    header_layout.setSpacing(8)

    icon = QLabel(_STATUS_ICON[status])
    icon.setFixedWidth(16)
    icon.setStyleSheet(
        f"color: {_STATUS_COLOR[status]}; font-family: sans-serif; font-weight: bold; font-size: 14px;"
    )
    header_layout.addWidget(icon)

    label_widget = QLabel(label)
    label_widget.setStyleSheet("color: white; font-family: sans-serif; font-size: 13px;")
    header_layout.addWidget(label_widget)
    header_layout.addStretch()
    layout.addWidget(header)

    if status != STATUS_OK:
        detail = QLabel(message)
        detail.setWordWrap(True)
        detail.setStyleSheet("color: #aaaaaa; font-family: sans-serif; font-size: 11px;")
        layout.addWidget(detail)

    return item


class LauncherWindow(QWidget):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("Dota Overlay — запуск")
        self.setFixedWidth(420)

        self._panel = _GradientPanel()
        self._panel_layout = QVBoxLayout(self._panel)
        self._panel_layout.setContentsMargins(18, 16, 18, 16)
        self._panel_layout.setSpacing(4)

        outer = QVBoxLayout(self)
        outer.setContentsMargins(0, 0, 0, 0)
        outer.addWidget(self._panel)

        title = QLabel("DOTA OVERLAY")
        title.setStyleSheet(
            "color: white; font-weight: bold; font-family: sans-serif; "
            "font-size: 14px; letter-spacing: 2px;"
        )
        self._panel_layout.addWidget(title)

        self._checks_container = QWidget()
        self._checks_layout = QVBoxLayout(self._checks_container)
        self._checks_layout.setContentsMargins(0, 8, 0, 0)
        self._checks_layout.setSpacing(0)
        self._panel_layout.addWidget(self._checks_container)

        buttons = QWidget()
        button_layout = QHBoxLayout(buttons)
        button_layout.setContentsMargins(0, 10, 0, 0)
        recheck_btn = QPushButton("Перепроверить")
        recheck_btn.clicked.connect(self.run_checks)
        self._launch_btn = QPushButton("Запустить")
        self._launch_btn.clicked.connect(self._on_launch)
        button_layout.addWidget(recheck_btn)
        button_layout.addWidget(self._launch_btn)
        self._panel_layout.addWidget(buttons)

        self.run_checks()

    def run_checks(self):
        while self._checks_layout.count():
            item = self._checks_layout.takeAt(0)
            widget = item.widget()
            if widget:
                widget.deleteLater()

        has_error = False
        for label, fn in CHECKS:
            status, message = fn()
            has_error = has_error or status == STATUS_ERROR
            self._checks_layout.addWidget(_check_item(label, status, message))

        self._launch_btn.setEnabled(not has_error)
        self._panel.adjustSize()
        self.adjustSize()

    def _on_launch(self):
        subprocess.Popen([sys.executable, "app.py"], cwd=PROJECT_DIR)
        self.close()
        QApplication.instance().quit()


if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = LauncherWindow()
    window.show()
    sys.exit(app.exec())
```
